cmake_minimum_required(VERSION 3.20)
project(AegisGate LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(EXISTS "${CMAKE_SOURCE_DIR}/.deps/boost_1_90_0/boost/beast.hpp")
  set(BOOST_ROOT "${CMAKE_SOURCE_DIR}/.deps/boost_1_90_0")
endif()
find_path(BOOST_INCLUDE boost/beast.hpp HINTS ${BOOST_ROOT} REQUIRED)
file(GLOB PROXY_SOURCES CONFIGURE_DEPENDS proxy/src/*.cpp)
add_executable(aegis-proxy ${PROXY_SOURCES})
target_include_directories(aegis-proxy PRIVATE proxy/include ${BOOST_INCLUDE})
target_compile_definitions(aegis-proxy PRIVATE BOOST_ERROR_CODE_HEADER_ONLY)
if(WIN32)
  target_link_libraries(aegis-proxy PRIVATE ws2_32 mswsock)
else()
  find_package(Threads REQUIRED)
  target_link_libraries(aegis-proxy PRIVATE Threads::Threads)
endif()
enable_testing()
add_executable(core-tests proxy/tests/core.cpp proxy/src/ip.cpp proxy/src/cidr.cpp)
target_include_directories(core-tests PRIVATE proxy/include)
add_test(NAME core COMMAND core-tests)
